# ai-adapter status / uninstall コマンド実装指示書

> **対象コマンド**: `ai-adapter status`, `ai-adapter uninstall`  
> **対応フェーズ**: フェーズ 3（基盤機能）

## 1. コマンド概要

| コマンド | ステータス | 説明 |
|---------|-----------|------|
| `ai-adapter status` | ✅ 実装済み（`src/ai_adapter/cli.py`） | 現在の設定状態を表示する |
| `ai-adapter uninstall` | ❌ 未実装 | `~/.ai-adapter/` を削除して初期状態に戻す |

## 2. `ai-adapter status` の実装詳細

### 2.1 現在の実装（`src/ai_adapter/cli.py`）

`status` コマンドは既に `cli.py` のトップレベルコマンドとして実装されている。

**ファイル**: `src/ai_adapter/cli.py`（`cmd_status` 関数）

```python
@main.command(name="status")
def cmd_status() -> None:
    """現在の状態を表示する。"""
    adapter_dir = _config.AI_ADAPTER_DIR
    if not adapter_dir.exists():
        click.echo("ai-adapter は初期化されていません。")
        click.echo("ai-adapter init を実行してください。")
        return

    config = _config.load_config()
    if config is None:
        click.echo("設定ファイルが見つかりません。")
        click.echo("ai-adapter init を実行してください。")
        return

    click.echo("ai-adapter 状態:")
    click.echo(f"  データディレクトリ: {adapter_dir}")
    click.echo(f"  設定ファイル: {_config.get_config_path()}")
    click.echo(f"  バージョン: {config.version}")
    click.echo(f"  デフォルト環境: {config.default_env}")
    click.echo(f"  登録エージェント数: {len(config.agents)}")
    click.echo(f"  登録環境数: {len(config.envs)}")
    click.echo(f"  登録スクリプト数: {len(config.bins)}")
    click.echo(f"  登録スキル数: {len(config.skills)}")
    click.echo(f"  MCP サーバー数: {len(config.mcp_servers)}")
    click.echo(f"  エージェント紐付け数: {len(config.agent_bindings)}")

    # ディレクトリ存在確認
    agents_dir = adapter_dir / "agents"
    bins_dir = adapter_dir / "bin"
    skills_dir = adapter_dir / "skills"
    mcp_dir = adapter_dir / "mcp"
    click.echo(f"  agents/ ディレクトリ: {'✓' if agents_dir.exists() else '✗'}")
    click.echo(f"  bin/ ディレクトリ: {'✓' if bins_dir.exists() else '✗'}")
    click.echo(f"  skills/ ディレクトリ: {'✓' if skills_dir.exists() else '✗'}")
    click.echo(f"  mcp/ ディレクトリ: {'✓' if mcp_dir.exists() else '✗'}")
```


### 2.3 注意点

- `status` は **`cli.py` のトップレベル** に実装されている（サブコマンドグループではない）
- 未初期化時、設定ファイル不在時はそれぞれ適切なメッセージを表示して終了する
- ディレクトリの存在確認は `exists()` で行い、`✓` / `✗` で表示する
- `config.skills` / `config.mcp_servers` は models.py で定義されていれば表示する
  - 未定義の場合は `getattr` で安全に取得するか、models.py を先に更新する

## 3. `ai-adapter uninstall` の実装詳細

### 3.1 処理フロー

```
ai-adapter uninstall
  │
  ├─ Step 1: ~/.ai-adapter/ が存在するか確認
  │   ├─ Yes → そのまま
  │   └─ No  → 「ai-adapter は初期化されていません」と表示して終了
  │
  ├─ Step 2: 確認プロンプトの表示
  │   ├─ Yes → 削除を続行
  │   └─ No  → 処理を中断
  │
  ├─ Step 3: Git リポジトリの場合の警告
  │   └─ リモートが設定されていれば変更のプッシュを促す
  │
  ├─ Step 4: ~/.ai-adapter/ を丸ごと削除
  │
  └─ Step 5: 完了メッセージを表示
```

### 3.2 データ削除の設計

**削除対象:**

```
~/.ai-adapter/          ← 丸ごと削除（shutil.rmtree）
├── config.json         ← 設定ファイル
├── agents/             ← 登録エージェントファイル
├── bin/                ← 登録スクリプトファイル
├── skills/             ← 登録スキル
└── mcp/                ← MCPサーバー設定
```

**削除しないもの:**

- `.github/agents/` / `.github/bin/`（プロジェクトに展開済みのファイル）
- `.claude/skills/`（プロジェクトに展開済みのファイル）
- `.vscode/mcp.json` / `.mcp.json`（エクスポート済みの MCP 設定）

### 3.3 実装コード（`src/ai_adapter/cli.py` に追加）

```python
import shutil
from pathlib import Path
import click
from ai_adapter import config as _config


@main.command(name="uninstall")
@click.option("--force", is_flag=True, help="確認プロンプトを表示せずに削除する")
@click.option("--keep-git", is_flag=True, help="Git リポジトリ情報を保持する")
def cmd_uninstall(force: bool, keep_git: bool) -> None:
    """~/.ai-adapter/ を削除して初期状態に戻す。"""
    adapter_dir = _config.AI_ADAPTER_DIR

    if not adapter_dir.exists():
        click.echo("ai-adapter は初期化されていません。")
        click.echo("削除するデータはありません。")
        return

    # Git リポジトリ確認
    git_dir = adapter_dir / ".git"
    is_git_repo = git_dir.exists()

    if is_git_repo and not keep_git:
        click.echo("警告: ~/.ai-adapter/ は Git リポジトリとして管理されています。")
        click.echo("リモートの変更を先にプッシュすることを推奨します。")
        click.echo("  cd ~/.ai-adapter && git status")
        click.echo("  git push")

    # 確認プロンプト
    if not force:
        size_info = _get_dir_size(adapter_dir)
        click.echo(f"削除対象: {adapter_dir} ({size_info})")
        click.confirm("本当に削除しますか？", abort=True)

    # ~/.ai-adapter/ を削除
    if keep_git and is_git_repo:
        # Git リポジトリは保持し、データのみ削除
        _remove_contents_except_git(adapter_dir)
        click.echo(f"データを削除しました (Git リポジトリは保持): {adapter_dir}")
    else:
        shutil.rmtree(adapter_dir)
        click.echo(f"アンインストールしました: {adapter_dir}")

    click.echo("ai-adapter init で再初期化できます。")


def _get_dir_size(path: Path) -> str:
    """ディレクトリのサイズを人間が読める形式で返す。"""
    total = 0
    for f in path.rglob("*"):
        if f.is_file():
            total += f.stat().st_size
    if total < 1024:
        return f"{total} B"
    elif total < 1024 * 1024:
        return f"{total / 1024:.1f} KB"
    else:
        return f"{total / 1024 / 1024:.1f} MB"


def _remove_contents_except_git(path: Path) -> None:
    """Git リポジトリ（.git）を残して中身をすべて削除する。"""
    for item in path.iterdir():
        if item.name == ".git":
            continue
        if item.is_dir():
            shutil.rmtree(item)
        else:
            item.unlink()
```

```python
# tests/test_cli.py に既存（確認用）
def test_init_and_status(self):
    """init → status の流れを確認する。"""
    runner = CliRunner()
    with runner.isolated_filesystem():
        with mock.patch("pathlib.Path.home", return_value=Path(".")):
            # init
            result = runner.invoke(main, ["init"])
            self.assertEqual(result.exit_code, 0)
            # status
            result = runner.invoke(main, ["status"])
            self.assertEqual(result.exit_code, 0)
            self.assertIn("ai-adapter 状態:", result.output)
            self.assertIn("登録エージェント数", result.output)


def test_status_before_init(self):
    """init 前の status で適切なメッセージが表示されることを確認する。"""
    runner = CliRunner()
    with runner.isolated_filesystem():
        with mock.patch("pathlib.Path.home", return_value=Path(".")):
            result = runner.invoke(main, ["status"])
            self.assertIn("初期化されていません", result.output)
```

### 5.2 `uninstall` のテスト（`test_cli.py` に追加）

```python
# tests/test_cli.py に追加
class TestUninstallCommand(unittest.TestCase):
    """uninstall コマンドのテスト。"""

    def test_uninstall_before_init(self):
        """未初期化時に uninstall するとメッセージが表示されることを確認する。"""
        runner = CliRunner()
        with runner.isolated_filesystem():
            with mock.patch("pathlib.Path.home", return_value=Path(".")):
                result = runner.invoke(main, ["uninstall", "--force"])
                self.assertEqual(result.exit_code, 0)
                self.assertIn("初期化されていません", result.output)

    def test_uninstall_after_init(self):
        """init → uninstall の流れを確認する。"""
        runner = CliRunner()
        with runner.isolated_filesystem():
            with mock.patch("pathlib.Path.home", return_value=Path(".")):
                runner.invoke(main, ["init"])
                self.assertTrue(Path(".ai-adapter").exists())

                result = runner.invoke(main, ["uninstall", "--force"])
                self.assertEqual(result.exit_code, 0)
                self.assertIn("アンインストールしました", result.output)
                self.assertFalse(Path(".ai-adapter").exists())

    def test_uninstall_keep_git(self):
        """--keep-git で .git が保持されることを確認する。"""
        runner = CliRunner()
        with runner.isolated_filesystem():
            with mock.patch("pathlib.Path.home", return_value=Path(".")):
                runner.invoke(main, ["init"])
                # Git リポジトリを模擬
                git_dir = Path(".ai-adapter/.git")
                git_dir.mkdir(parents=True, exist_ok=True)
                (git_dir / "HEAD").write_text("ref: refs/heads/main\n")

                result = runner.invoke(main, ["uninstall", "--force", "--keep-git"])
                self.assertEqual(result.exit_code, 0)
                self.assertIn("Git リポジトリは保持", result.output)
                self.assertTrue(Path(".ai-adapter/.git").exists())
                self.assertFalse(Path(".ai-adapter/config.json").exists())

    def test_uninstall_cancel(self):
        """確認プロンプトで No を選択すると削除されないことを確認する。"""
        runner = CliRunner()
        with runner.isolated_filesystem():
            with mock.patch("pathlib.Path.home", return_value=Path(".")):
                runner.invoke(main, ["init"])
                result = runner.invoke(main, ["uninstall"], input="n\n")
                self.assertEqual(result.exit_code, 0)
                self.assertTrue(Path(".ai-adapter").exists())
```
