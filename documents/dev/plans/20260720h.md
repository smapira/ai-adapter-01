# ai-adapter init / start / sync 改修指示書

> **作成年月日**: 2026-07-20  
> **対象コマンド**: `ai-adapter init`, `ai-adapter start`, `ai-adapter sync`  
> **目的**: Git リモート設定の手動操作を廃し、対話的セットアップに統一する

---

## 1. 現状の問題点

現在の `ai-adapter` では、GitHub との同期を利用するために **ユーザーが手動で** 以下を実行する必要がある:

```bash
cd ~/.ai-adapter
git remote add origin git@github.com:user/my-agent-config.git
```

この手動ステップを廃し、以下のように **コマンド内で完結** させる:

| シナリオ | 現状 | 改善後 |
|---------|------|--------|
| 新規セットアップ | `ai-adapter init` → 手動で `git remote add` | `ai-adapter start <URL>` 一発 |
| 対話的セットアップ | なし | `ai-adapter init` がリモートURLを対話的に質問 |
| 既存リポジトリのクローン | `git clone <url> ~/.ai-adapter`（手動） | `ai-adapter init` が自動クローン |

## 2. `ai-adapter start <URL>` — 新規コマンド

### 2.1 処理フロー

```
ai-adapter start <URL>
  │
  ├─ Step 1: ~/.ai-adapter/ が存在するか確認
  │   ├─ 存在しない → OK、そのまま進む
  │   └─ 存在する → 「既に初期化されています。上書きしますか？」と確認
  │
  ├─ Step 2: リモートURLを config.json に保存
  │   └─ config.json は後で自動生成される
  │
  ├─ Step 3: git clone <URL> ~/.ai-adapter/
  │   ├─ 成功 → クローン完了、既存の設定を引き継ぐ
  │   └─ 失敗（空リポジトリ）→ git init + git remote add で初期化
  │
  ├─ Step 4: 初期化（初回のみ）
  │   └─ ~/.ai-adapter/ に agents/ bin/ skills/ mcp/ を作成
  │   └─ config.json が存在しなければデフォルト設定を生成
  │
  └─ Step 5: 完了メッセージ
```

### 2.2 シグネチャ

```python
@main.command(name="start")
@click.argument("url")
def cmd_start(url: str) -> None:
    """GitHub リモートと連携して ~/.ai-adapter/ を初期化する。

    URL: Git リモートリポジトリのURL（例: git@github.com:user/my-agent-config.git）
    """
```

### 2.3 実装詳細

```python
from ai_adapter.git import GitError, init_repo, is_repo, has_remote, get_remotes
from ai_adapter.git import _run_git as run_git


@main.command(name="start")
@click.argument("url")
def cmd_start(url: str) -> None:
    """GitHub リモートと連携して ~/.ai-adapter/ を初期化する。"""
    adapter_dir = _config.AI_ADAPTER_DIR

    if adapter_dir.exists():
        click.echo(f"'{adapter_dir}' は既に存在します。")
        click.confirm("既存の設定を上書きしますか？（設定はマージされます）", abort=True)

    # Step 1: git clone を試みる
    click.echo(f"リモートリポジトリからクローン中: {url}")
    try:
        parent = adapter_dir.parent
        run_git(["clone", url, str(adapter_dir.name)], cwd=parent)
        click.echo("クローンしました。")
    except GitError as e:
        # 空リポジトリ or 初回 → git init + remote add
        click.echo("クローンに失敗しました。新規リポジトリとして初期化します。")
        adapter_dir.mkdir(parents=True, exist_ok=True)
        init_repo(adapter_dir)
        run_git(["remote", "add", "origin", url], cwd=adapter_dir)
        click.echo(f"リモートを設定しました: {url}")

    # Step 2: ディレクトリ構造を初期化
    dirs = [
        adapter_dir / "agents",
        adapter_dir / "bin",
        adapter_dir / "skills",
        adapter_dir / "mcp",
    ]
    for d in dirs:
        d.mkdir(parents=True, exist_ok=True)

    # Step 3: config.json がなければデフォルト生成
    config_path = adapter_dir / "config.json"
    if not config_path.exists():
        config = Config(
            version=1,
            default_env="default",
            envs=[Env(name="default", description="デフォルト環境")],
            agent_bindings=[],
        )
        save_config(config)
        click.echo("デフォルト設定ファイルを生成しました。")

    click.echo(f"セットアップ完了: {adapter_dir}")
    click.echo(f"リモート: {url}")
    click.echo("ai-adapter sync で設定を同期できます。")
```

## 3. `ai-adapter init` — 対話的プロンプト追加

### 3.1 現状の動作

```python
@main.command(name="init")
def cmd_init() -> None:
    """~/.ai-adapter/ を初期化する。"""
    created = _config.init()
    if created:
        click.echo(f"初期化しました: {_config.AI_ADAPTER_DIR}")
```

### 3.2 改善後の動作

`init` 実行時、リモートリポジトリが未設定の場合に **対話的にURLを質問** する。

```python
@main.command(name="init")
@click.option("--remote", "-r", help="Git リモートリポジトリのURL")
def cmd_init(remote: str | None) -> None:
    """~/.ai-adapter/ を初期化する。

    リモートURLが未指定の場合、対話的に入力を促します。
    """
    created = _config.init()
    if created:
        click.echo(f"初期化しました: {_config.AI_ADAPTER_DIR}")
        click.echo(f"設定ファイル: {_config.get_config_path()}")
    else:
        click.echo(f"既に初期化されています: {_config.AI_ADAPTER_DIR}")

    # リモート設定
    adapter_dir = _config.AI_ADAPTER_DIR
    from ai_adapter.git import has_remote, init_repo, is_repo, _run_git as run_git

    if not is_repo(adapter_dir):
        init_repo(adapter_dir)

    if not has_remote(adapter_dir):
        # コマンドラインオプションがなければ対話的に質問
        if remote is None:
            click.echo()
            click.echo("--- GitHub 同期の設定 ---")
            click.echo("設定を複数PCで共有するには、GitHub リポジトリのURLを入力してください。")
            click.echo("（スキップする場合は何も入力せず Enter を押してください）")
            remote = click.prompt(
                "Git リモートリポジトリURL",
                default="",
                show_default=False,
            ).strip()

        if remote:
            run_git(["remote", "add", "origin", remote], cwd=adapter_dir)
            click.echo(f"リモートを設定しました: {remote}")
            click.echo("ai-adapter sync で設定を同期できます。")
        else:
            click.echo("リモートは設定されませんでした。")
            click.echo("後から ai-adapter start <URL> で設定することもできます。")
    else:
        remotes = get_remotes(adapter_dir)
        click.echo(f"リモートは既に設定されています: {', '.join(remotes)}")
```

## 4. `ai-adapter sync` — リモート未設定時のエラーハンドリング改善

### 4.1 現状の問題

現在の `sync` はリモート未設定時に以下のメッセージを出して **エラー終了** する:

```
リモートリポジトリが設定されていません。
git remote add origin <repository-url> を実行してください。
```

### 4.2 改善後

リモート未設定の場合、`init` と同様に対話的に入力を促す:

```python
# sync.py の変更箇所
if not has_remote(adapter_dir):
    click.echo()
    click.echo("リモートリポジトリが設定されていません。")

    # すでに `start` で config.json に remote が保存されているか確認
    config = _config.load_config()
    saved_remote = getattr(config, "remote", None) if config else None

    if saved_remote:
        click.echo(f"保存されたリモートURLが見つかりました: {saved_remote}")
        run_git(["remote", "add", "origin", saved_remote], cwd=adapter_dir)
        click.echo("リモートを設定しました。")
    else:
        remote = click.prompt(
            "Git リモートリポジトリURL（スキップするには Enter）",
            default="",
            show_default=False,
        ).strip()

        if not remote:
            click.echo("リモートが設定されていないため、同期をスキップします。")
            click.echo("ai-adapter start <URL> で後から設定できます。")
            return

        run_git(["remote", "add", "origin", remote], cwd=adapter_dir)
        click.echo(f"リモートを設定しました: {remote}")
```

## 6. git.py に `clone` / `add_remote` 関数を追加

### 6.1 追加する関数

```python
def clone(url: str, dest: Path) -> None:
    """git clone を実行する。"""
    _run_git(["clone", url, str(dest)], cwd=dest.parent)


def add_remote(path: Path, name: str, url: str) -> None:
    """git remote add を実行する。"""
    _run_git(["remote", "add", name, url], cwd=path)


def get_current_branch(path: Path) -> str:
    """現在のブランチ名を取得する。"""
    result = _run_git(["rev-parse", "--abbrev-ref", "HEAD"], cwd=path)
    return result.stdout.strip()
```

## 8. テスト方針

### 8.1 `test_start`（新規）

```python
class TestStartCommand(unittest.TestCase):
    """start コマンドのテスト"""

    def test_start_new_repo(self):
        """新規リポジトリの場合、init + remote add される"""
        with CliRunner().isolated_filesystem():
            with mock.patch("pathlib.Path.home", return_value=Path(".")):
                result = runner.invoke(main, ["start", "git@github.com:user/test.git"])
                self.assertEqual(result.exit_code, 0)
                self.assertIn("セットアップ完了", result.output)
                self.assertTrue(Path(".ai-adapter").exists())
                self.assertTrue(Path(".ai-adapter/agents").exists())
                self.assertTrue(Path(".ai-adapter/bin").exists())
                self.assertTrue(Path(".ai-adapter/config.json").exists())

    def test_start_existing(self):
        """既に存在する場合の確認プロンプト"""
        adapter_dir = Path(".ai-adapter")
        adapter_dir.mkdir(parents=True)
        result = runner.invoke(main, ["start", "git@github.com:user/test.git"], input="y\n")
        self.assertEqual(result.exit_code, 0)

    def test_start_with_remote_in_config(self):
        """リモートURLが config.json に保存される"""
        # init状態で start → config.json に remote が保存される
        result = runner.invoke(main, ["start", "git@github.com:user/test.git"])
        config = json.loads(Path(".ai-adapter/config.json").read_text())
        self.assertEqual(config.get("remote"), "git@github.com:user/test.git")

    def test_start_and_sync(self):
        """start → sync がエラーなく実行される"""
        result = runner.invoke(main, ["start", "git@github.com:user/test.git"])
        # sync はリモートが必要なので start 後に実行できる
        result = runner.invoke(main, ["sync"])
        self.assertIn("Step 1", result.output)
```
